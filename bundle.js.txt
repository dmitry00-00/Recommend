  function seedOf(s) {
    var n = 0;
    for (var i = 0; i < s.length; i++) n = (n * 31 + s.charCodeAt(i)) % 100000;
    return n;
  }

  // WorkCover — генеративная типографская обложка. Реальные постеры не
  // используются: изолинии строятся из названия, полоса кромки — из типа медиа.
  function WorkCover(p) {
    var work = p.work;
    var size = p.size || 'md';
    var seed = seedOf(work.title + work.year);
    var op = (work.primaryOperations && work.primaryOperations[0] && work.primaryOperations[0].op) || 'synthesis';
    var W0 = size === 'lg' ? 196 : size === 'sm' ? 84 : 124;
    var H0 = Math.round(W0 * (size === 'sm' ? 1.26 : 1.3));
    var holes = [];
    if (size === 'lg') for (var k = 0; k < 9; k++) holes.push(h('span', { key: k, className: 'tm-cover__hole' }));
    return h('div', {
      className: cx('tm-cover', 'tm-cover--' + size, 'tm-cover--' + work.type, p.className),
      style: { width: W0, height: H0, '--cover-line': opVar(op) },
      role: 'img', 'aria-label': work.title + ', ' + work.year
    },
      size === 'lg' ? h('span', { className: 'tm-cover__edge', 'aria-hidden': 'true' }, holes) : null,
      h('span', { className: 'tm-cover__title' }, work.title),
      h('span', { className: 'tm-cover__foot' },
        h('span', null, work.year),
        size !== 'sm' ? h('span', { className: 'tm-cover__frameno' }, (seed % 90 + 10) + ((seed % 3) ? 'A' : 'B')) : null)
    );
  }

  function UncertaintyMark(p) {

      h('div', { className: 'tm-unc__track' },

  function polar(cx0, cy0, r, i, n) {
    var a = (-90 + i * (360 / n)) * Math.PI / 180;
    return [cx0 + r * Math.cos(a), cy0 + r * Math.sin(a)];
  }
  function ring(cx0, cy0, r, n) {
    var pts = [];
    for (var i = 0; i < n; i++) pts.push(polar(cx0, cy0, r, i, n).map(function (v) { return Math.round(v * 10) / 10; }).join(','));
    return pts.join(' ');
  }

  function CognitiveMap(p) {

      (t.operations || []).forEach(function (o) { traceCount[o.op] = (traceCount[o.op] || 0) + 1; });
    });

    var table = h('table', { className: cx('tm-maplist', mode === 'field' && 'tm-sr') },
      h('caption', null, 'Карта мышления: уровень и диапазон по восьми операциям'),
      h('thead', null, h('tr', null,
        h('th', { scope: 'col' }, 'Операция'), h('th', { scope: 'col' }, 'Уровень'),
        h('th', { scope: 'col' }, 'Диапазон'), h('th', { scope: 'col' }, 'Данных'))),
      h('tbody', null, ops.map(function (o) {
        var meta = OPS[o.op];
        return h('tr', { key: o.op, className: targets[o.op] ? 'tm-maplist__row--target' : null },

          h('td', null, p.showDetails ? o.level.toFixed(1) : ['совсем немного', 'немного', 'средне', 'уверенно', 'очень уверенно'][Math.min(4, Math.floor(o.level / 2.5))]),

          : 'Сплошная линия — оценка, штриховка — её диапазон. Короткие засечки на осях — пройденные произведения.'),

  function AssessmentItem(p) {
    var item = p.item, r = item.response;
    var st = React.useState(r.type === 'ordering' ? r.items.slice() : (r.type === 'multi_choice' ? [] : null));
    var val = st[0], setVal = st[1];
    var grid = React.useState({}), fam = grid[0], setFam = grid[1];

    function move(i, dir) {
      var next = val.slice(), j = i + dir;
      if (j < 0 || j >= next.length) return;
      var tmp = next[i]; next[i] = next[j]; next[j] = tmp;
      setVal(next);
    }

    var body;
    if (r.type === 'familiarity_grid') {
      body = h('div', { className: 'tm-agrid' }, r.works.map(function (w2) {
        return h('div', { key: w2.id, className: 'tm-agrid__cell' },
          h(WorkCover, { work: w2, size: 'sm' }),
          h('p', { className: 'tm-agrid__title' }, w2.title),
          h('div', { className: 'tm-agrid__levels', role: 'radiogroup', 'aria-label': w2.title },
            FAMILIARITY.map(function (f) {
              var on = fam[w2.id] === f.id;
              return h('button', { key: f.id, type: 'button', role: 'radio', 'aria-checked': on ? 'true' : 'false',
                className: cx('tm-agrid__lvl', on && 'tm-agrid__lvl--on'),
                onClick: function () { var n = Object.assign({}, fam); n[w2.id] = f.id; setFam(n); } }, f.label);
            })));
      }));
    } else if (r.type === 'single_choice' || r.type === 'multi_choice') {
      var multi = r.type === 'multi_choice';
      body = h('div', { className: 'tm-achoice' }, r.options.map(function (o) {
        var on = multi ? (val || []).indexOf(o.id) >= 0 : val === o.id;
        return h('button', { key: o.id, type: 'button', className: cx('tm-achoice__opt', on && 'tm-achoice__opt--on'),
          'aria-pressed': on ? 'true' : 'false',
          onClick: function () {
            if (!multi) return setVal(o.id);
            var n = (val || []).slice(), i = n.indexOf(o.id);
            if (i >= 0) n.splice(i, 1); else n.push(o.id);
            setVal(n);
          } },
          h('span', { className: 'tm-achoice__mark', 'aria-hidden': 'true' }), o.label);
      }));
    } else if (r.type === 'ordering') {
      body = h('ol', { className: 'tm-aorder' }, val.map(function (it, i) {
        return h('li', { key: it.id, className: 'tm-aorder__row' },
          h('span', { className: 'tm-aorder__n' }, i + 1),
          h('span', { className: 'tm-aorder__label' }, it.label),
          h('span', { className: 'tm-aorder__btns' },
            h('button', { type: 'button', className: 'tm-aorder__btn', 'aria-label': 'Выше: ' + it.label,
              disabled: i === 0, onClick: function () { move(i, -1); } }, '↑'),
            h('button', { type: 'button', className: 'tm-aorder__btn', 'aria-label': 'Ниже: ' + it.label,
              disabled: i === val.length - 1, onClick: function () { move(i, 1); } }, '↓')));
      }));
    } else if (r.type === 'scale') {
      var steps = [];
      for (var i = r.min; i <= r.max; i++) {
        (function (v) {
          steps.push(h('button', { key: v, type: 'button', role: 'radio', 'aria-checked': val === v ? 'true' : 'false',
            'aria-label': 'позиция ' + v + ' из ' + r.max,
            className: cx('tm-ascale__p', val === v && 'tm-ascale__p--on'), onClick: function () { setVal(v); } }));
        })(i);
      }
      body = h('div', { className: 'tm-ascale' },
        h('div', { className: 'tm-ascale__row', role: 'radiogroup', 'aria-label': item.prompt }, steps),
        h(Row, { className: 'tm-ascale__ends' },
          h('span', null, r.minLabel), h('span', null, r.maxLabel)));
    } else {
      body = h('div', null,
        h('textarea', { className: 'tm-refl__area', rows: 4, maxLength: r.maxLength,
          placeholder: 'Пара предложений. Можно пропустить.', value: val || '',
          onChange: function (e) { setVal(e.target.value); } }),
        h('p', { className: 'tm-refl__count' }, (val || '').length + ' / ' + r.maxLength));
    }

    return h('section', { className: 'tm-aitem' },
      p.progress !== false ? h(AssessmentProgress, p.progress || { answered: 3, estimatedTotal: 8, minutesLeft: 6 }) : null,
      h('h2', { className: 'tm-aitem__prompt' }, item.prompt),
      item.body ? h('p', { className: 'tm-aitem__body' }, item.body) : null,
      body);
  }
  C.AssessmentItem = AssessmentItem;

  function AgreementMatrix(p) {
    var fields = [];
    (p.rows || []).forEach(function (r) { if (fields.indexOf(r.field) < 0) fields.push(r.field); });
    var groups = ['experts', 'community', 'all'];
    var GL = { experts: 'Эксперты', community: 'Сообщество', all: 'Все' };
    return h('div', { className: 'tm-table__wrap' },
      h('table', { className: 'tm-table tm-table--matrix' },
        h('caption', null, p.caption || 'Согласованность по полям, α Криппендорфа'),
        h('thead', null, h('tr', null, h('th', { scope: 'col' }, 'Поле'),
          groups.map(function (g) { return h('th', { key: g, scope: 'col' }, GL[g]); }))),
        h('tbody', null, fields.map(function (f) {
          return h('tr', { key: f },
            h('th', { scope: 'row' }, h('code', null, f)),
            groups.map(function (g) {
              var cell = (p.rows || []).filter(function (r) { return r.field === f && r.raterGroup === g; })[0];
              if (!cell) return h('td', { key: g, className: 'tm-table__num' }, '—');
              var a = AGREE[cell.status];
              return h('td', { key: g, className: 'tm-table__num' },
                h('span', { className: cx('tm-alpha', 'tm-alpha--' + a.tone) }, cell.alpha.toFixed(2)),
                h('span', { className: 'tm-table__sub' }, a.label + ' · ' + cell.raters));
            }));
        }))),
      p.example ? h('p', { className: 'tm-matrix__example' }, p.example) : null);
  }
